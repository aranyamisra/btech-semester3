;POLLING METHOD

ORG 0000h

CLR P0.7

UP:
MOV TMOD, #01H	;TIMER 0 IN MODE 1
MOV TH0, #0FFH	;LOAD COUNT
MOV TL0, #0F0H
MOV P1, A
SETB TCON.4		;START TIMER
HERE: JNB TCON.5, HERE		;POLLING
CLR TCON.4		;STOP TIMER
CLR TCON.5		;CLEAR TIMER OVERFLOW
CPL A			;COMPLEMENT A
SJMP UP
